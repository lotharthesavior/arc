1//! # SQLite Read Model Store
2//!
3//! Production-grade implementation of [`ReadModelStore`] backed by an
4//! `r2d2`-pooled SQLite connection. Each projection table follows the
5//! framework's standard projection shape: `(id TEXT PK, version BIGINT,
6//! data TEXT)`. The `data` column holds the full row as JSON, so this store
7//! can serve any projector without compile-time knowledge of the
8//! domain-specific columns.
9//!
10//! ## Idempotency
11//!
12//! [`upsert`](ReadModelStore::upsert) translates to
13//!
14//! ```sql
15//! INSERT INTO {table} (id, version, data) VALUES (?, ?, ?)
16//! ON CONFLICT(id) DO UPDATE
17//!    SET version = excluded.version, data = excluded.data
18//!  WHERE {table}.version < excluded.version
19//! ```
20//!
21//! That gate makes replay-from-zero deterministic and tolerates at-least-once
22//! delivery: applying an older event twice, or out-of-order, never regresses
23//! state.
24//!
25//! ## Queries
26//!
27//! [`find_by`](ReadModelStore::find_by) uses `json_extract(data, '$.{field}')`
28//! to reach into the blob. Columns commonly queried (e.g. `email`) get
29//! expression indexes in the migration that creates the table.
30//!
31//! ## Boundaries
32//!
33//! The store does **not** create tables — DDL lives in `migrations/`. That
34//! keeps the production write path away from connection-time mutations and
35//! lets `diesel migration run` control schema evolution.
36

45type Pool = r2d2::Pool<ConnectionManager<SqliteConnection>>;

47/// SQLite implementation of [`ReadModelStore`].
48#[derive(Clone)]
49pub struct SqliteReadModelStore {
50    pool: Arc<Pool>,
51}
52
53impl SqliteReadModelStore {
54    /// Build a new store from a database URL. Creates a small r2d2 pool.
55    pub async fn new(database_url: &str) -> ReadModelResult<Self> {
56        let manager = ConnectionManager::<SqliteConnection>::new(database_url);
57        let pool = Pool::builder().max_size(10).build(manager).map_err(|e| {
58            ReadModelError::other(format!("Failed to build read-model pool: {}", e))
59        })?;
60        Ok(SqliteReadModelStore {
61            pool: Arc::new(pool),
62        })
63    }
64
65    /// Build a store from an existing pool. Used in tests so we can share a
66    /// single in-memory database with the event store.
67    pub fn with_pool(pool: Pool) -> Self {
68        Self {
69            pool: Arc::new(pool),
70        }
71    }
72}

74#[derive(QueryableByName, Debug)]
75struct DataRow {
76    #[diesel(sql_type = Text)]
77    data: String,
78}
79
80/// Validate a table or column name against an allow-list of characters before
81/// splicing it into a SQL string. Diesel's `sql_query` does not bind
82/// identifiers, only values. This guards against caller-supplied identifiers
83/// reaching SQL with anything other than `[A-Za-z0-9_]` — same posture as
84/// every other SQL library that accepts dynamic table names.
85fn check_ident(label: &str, ident: &str) -> ReadModelResult<()> {
86    if ident.is_empty() {
87        return Err(ReadModelError::other(format!("{label} cannot be empty")));
88    }
89    if !ident.chars().all(|c| c.is_ascii_alphanumeric() || c == '_') {
90        return Err(ReadModelError::other(format!(
91            "invalid {label} '{ident}': only [A-Za-z0-9_] permitted"
92        )));
93    }
94    Ok(())
95}
96
97fn extract_version(row: &Row) -> ReadModelResult<i64> {
98    row.get("version").and_then(|v| v.as_i64()).ok_or_else(|| {
99        ReadModelError::write_failed(
100            "Upsert.row missing required i64 field 'version' for version-gated upsert",
101        )
102    })
103}
104
105fn parse_data_row(raw: &str) -> ReadModelResult<Row> {
106    serde_json::from_str(raw).map_err(|e| {
107        ReadModelError::query_failed(format!("Failed to parse projection row JSON: {e}"))
108    })
109}

111#[async_trait]
112impl ReadModelStore for SqliteReadModelStore {
113    async fn upsert(&self, op: Upsert) -> ReadModelResult<()> {
114        check_ident("table name", &op.table)?;
115        let version = extract_version(&op.row)?;
116        let data = serde_json::to_string(&op.row)
117            .map_err(|e| ReadModelError::write_failed(format!("Failed to serialize row: {e}")))?;
118        let pool = self.pool.clone();
119        let table = op.table.clone();
120        let key = op.key.clone();
121
122        tokio::task::spawn_blocking(move || -> ReadModelResult<()> {
123            let mut conn = pool.get().map_err(|e| {
124                ReadModelError::write_failed(format!("Failed to get connection: {e}"))
125            })?;
126
127            let sql = format!(
128                "INSERT INTO {table} (id, version, data) VALUES (?, ?, ?) \
129                 ON CONFLICT(id) DO UPDATE SET version = excluded.version, data = excluded.data \
130                 WHERE {table}.version < excluded.version"
131            );
132
133            diesel::sql_query(sql)
134                .bind::<Text, _>(key)
135                .bind::<BigInt, _>(version)
136                .bind::<Text, _>(data)
137                .execute(&mut *conn)
138                .map_err(|e| ReadModelError::write_failed(e.to_string()))?;
139            Ok(())
140        })
141        .await
142        .map_err(|e| ReadModelError::other(format!("Task join error: {e}")))?
143    }
144
145    async fn delete(&self, table: &str, key: &str) -> ReadModelResult<()> {
146        check_ident("table name", table)?;
147        let pool = self.pool.clone();
148        let table = table.to_string();
149        let key = key.to_string();
150
151        tokio::task::spawn_blocking(move || -> ReadModelResult<()> {
152            let mut conn = pool.get().map_err(|e| {
153                ReadModelError::write_failed(format!("Failed to get connection: {e}"))
154            })?;
155            diesel::sql_query(format!("DELETE FROM {table} WHERE id = ?"))
156                .bind::<Text, _>(key)
157                .execute(&mut *conn)
158                .map_err(|e| ReadModelError::write_failed(e.to_string()))?;
159            Ok(())
160        })
161        .await
162        .map_err(|e| ReadModelError::other(format!("Task join error: {e}")))?
163    }
164
165    async fn get(&self, table: &str, key: &str) -> ReadModelResult<Option<Row>> {
166        check_ident("table name", table)?;
167        let pool = self.pool.clone();
168        let table = table.to_string();
169        let key = key.to_string();
170
171        let raw = tokio::task::spawn_blocking(move || -> ReadModelResult<Option<String>> {
172            let mut conn = pool.get().map_err(|e| {
173                ReadModelError::query_failed(format!("Failed to get connection: {e}"))
174            })?;
175            let rows: Vec<DataRow> =
176                diesel::sql_query(format!("SELECT data FROM {table} WHERE id = ? LIMIT 1"))
177                    .bind::<Text, _>(key)
178                    .load(&mut *conn)
179                    .map_err(|e| ReadModelError::query_failed(e.to_string()))?;
180            Ok(rows.into_iter().next().map(|r| r.data))
181        })
182        .await
183        .map_err(|e| ReadModelError::other(format!("Task join error: {e}")))??;
184
185        match raw {
186            Some(s) => Ok(Some(parse_data_row(&s)?)),
187            None => Ok(None),
188        }
189    }
190
191    async fn find_by(
192        &self,
193        table: &str,
194        field: &str,
195        value: &serde_json::Value,
196    ) -> ReadModelResult<Vec<Row>> {
197        check_ident("table name", table)?;
198        check_ident("field name", field)?;
199        let pool = self.pool.clone();
200        let table = table.to_string();
201        let path = format!("$.{field}");
202
203        // We only support primitive comparisons here. Anything else is a
204        // caller bug, not a runtime exception worth tolerating.
205        let bind_text: Option<String> = match value {
206            serde_json::Value::Null => None,
207            serde_json::Value::String(s) => Some(s.clone()),
208            serde_json::Value::Number(n) => Some(n.to_string()),
209            serde_json::Value::Bool(b) => Some(if *b { "1".into() } else { "0".into() }),
210            other => {
211                return Err(ReadModelError::query_failed(format!(
212                    "find_by only accepts primitive values, got: {other}"
213                )))
214            }
215        };
216
217        let rows = tokio::task::spawn_blocking(move || -> ReadModelResult<Vec<String>> {
218            let mut conn = pool.get().map_err(|e| {
219                ReadModelError::query_failed(format!("Failed to get connection: {e}"))
220            })?;
221            let sql = format!("SELECT data FROM {table} WHERE json_extract(data, ?) IS ?");
222            let rows: Vec<DataRow> = diesel::sql_query(sql)
223                .bind::<Text, _>(path)
224                .bind::<Nullable<Text>, _>(bind_text)
225                .load(&mut *conn)
226                .map_err(|e| ReadModelError::query_failed(e.to_string()))?;
227            Ok(rows.into_iter().map(|r| r.data).collect())
228        })
229        .await
230        .map_err(|e| ReadModelError::other(format!("Task join error: {e}")))??;
231
232        rows.iter().map(|r| parse_data_row(r)).collect()
233    }
234
235    async fn list(&self, table: &str) -> ReadModelResult<Vec<Row>> {
236        check_ident("table name", table)?;
237        let pool = self.pool.clone();
238        let table = table.to_string();
239
240        let rows = tokio::task::spawn_blocking(move || -> ReadModelResult<Vec<String>> {
241            let mut conn = pool.get().map_err(|e| {
242                ReadModelError::query_failed(format!("Failed to get connection: {e}"))
243            })?;
244            let rows: Vec<DataRow> = diesel::sql_query(format!("SELECT data FROM {table}"))
245                .load(&mut *conn)
246                .map_err(|e| ReadModelError::query_failed(e.to_string()))?;
247            Ok(rows.into_iter().map(|r| r.data).collect())
248        })
249        .await
250        .map_err(|e| ReadModelError::other(format!("Task join error: {e}")))??;
251
252        rows.iter().map(|r| parse_data_row(r)).collect()
253    }
254
255    async fn truncate(&self, table: &str) -> ReadModelResult<()> {
256        check_ident("table name", table)?;
257        let pool = self.pool.clone();
258        let table = table.to_string();
259
260        tokio::task::spawn_blocking(move || -> ReadModelResult<()> {
261            let mut conn = pool.get().map_err(|e| {
262                ReadModelError::schema_failed(format!("Failed to get connection: {e}"))
263            })?;
264            diesel::sql_query(format!("DELETE FROM {table}"))
265                .execute(&mut *conn)
266                .map_err(|e| ReadModelError::schema_failed(e.to_string()))?;
267            Ok(())
268        })
269        .await
270        .map_err(|e| ReadModelError::other(format!("Task join error: {e}")))?
271    }
272}
